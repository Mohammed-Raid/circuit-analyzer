# Chaine de conditionnement multi-AOP (vue ILOT, comme chaine_5_aop) : flux OUT->IN strict.
# Etages mono-entree chaines en ligne : non-inverseur -> inverseur -> Schmitt -> comparateur
# Montre les 2 nouveaux montages chainables (Schmitt, comparateur) dans la vue chaine.

# --- Etage 1 : non-inverseur (VIN -> A) ---
U1   /VIN   /N1    /A      /VCC   /GND
Rfa  /N1    /A     100k
Rga  /N1    /GND   10k

# --- Etage 2 : inverseur (A -> B) ---
U2   /GND   /N2    /B      /VCC   /GND
Rinb /A     /N2    10k
Rfb  /N2    /B     100k

# --- Etage 3 : bascule de Schmitt (B -> C, hysteresis) ---
U3   /P3    /GND   /C      /VCC   /GND
Rinc /B     /P3    10k
Rfc  /C     /P3    100k

# --- Etage 4 : comparateur final (C -> D, seuil vs masse) ---
U4   /C     /GND   /D      /VCC   /GND
